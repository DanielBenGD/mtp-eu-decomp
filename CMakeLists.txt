cmake_minimum_required(VERSION 3.16)
project(mpt_eu_recomp C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_executable(mpt_pc pc/src/main.c pc/src/runtime.c)
target_include_directories(mpt_pc PRIVATE pc/include)
if(MSVC)
  target_compile_options(mpt_pc PRIVATE /W4)
else()
  target_compile_options(mpt_pc PRIVATE -Wall -Wextra -Wpedantic)
endif()
